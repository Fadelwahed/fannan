#!/bin/bash
set -u
W=/workspace/fannan-mimar-wt; O=$W/test-results/r12; R=/workspace/r5run.sh
clean(){ sed 's#/tmp/r5s#<scratch>#g; s#fannan-browser-fixture-[A-Za-z0-9]\{6\}#fannan-browser-fixture-XXXX#g'; }
$R bash -c 'node --version; node --test release/tests/*.test.mjs' 2>&1 | clean > $O/node.log; echo node=${PIPESTATUS[0]}
$R bash -c 'rm runtime-deps; node --test release/tests/*.test.mjs' 2>&1 | clean > $O/node-without-deps.log; echo nodeless=${PIPESTATUS[0]}
$R bash -c 'python3 --version; python3 -B release/tests/test_guard_tools.py -v' 2>&1 | clean > $O/python.log; echo py=${PIPESTATUS[0]}
$R bash -c 'cd revisions/r2 && node --test release/tests/*.test.mjs' 2>&1 | clean > $O/r2-node.log; echo r2node=${PIPESTATUS[0]}
$R bash -c 'cd revisions/r2 && python3 -B release/tests/test_guard_tools.py -v' 2>&1 | clean > $O/r2-python.log; echo r2py=${PIPESTATUS[0]}
$R bash -c 'python3 -B scripts/test_package_r3.py -v' 2>&1 | clean > $O/package-tests.log; echo pkgtests=${PIPESTATUS[0]}
(cd $W && python3 -B scripts/test_package_r3.py -v) > $O/package-tests-git.log 2>&1; echo pkgtestsgit=$?
$R bash -c 'python3 -B scripts/package-r3.py self-test' 2>&1 | clean > $O/package-self-test.log; echo selftest=${PIPESTATUS[0]}
if [ -x /usr/bin/node ] && /usr/bin/node --version | grep -q '^v20'; then $R bash -c 'export PATH=/usr/bin:/bin; node --version; node --test release/tests/*.test.mjs' 2>&1 | clean > $O/node20.log; echo node20=${PIPESTATUS[0]}; else echo node20=UNAVAILABLE; fi
$R bash -c 'f=0; m=0; s=0; p=0; i=0; for x in release/tools/*.mjs release/tests/*.mjs; do m=$((m+1)); echo "mjs $x"; node --check $x || f=$((f+1)); done; for x in release/tools/*.sh; do s=$((s+1)); bash -n $x || f=$((f+1)); done; for x in release/tools/*.internal; do i=$((i+1)); bash -n $x || f=$((f+1)); done; for x in release/tools/*.py release/tests/*.py scripts/*.py; do p=$((p+1)); python3 -B -c "import ast,sys; ast.parse(open(sys.argv[1]).read())" $x || f=$((f+1)); done; echo "syntax: mjs=$m sh=$s py=$p launcher.internal=$i failures=$f"; exit $f' 2>&1 | clean > $O/syntax.log; echo syntax=${PIPESTATUS[0]}
(cd $W && PATH=/home/box/.local/node-v22.23.3/bin:/usr/bin:/bin node release/tests/local-fixture.mjs) 2>&1 | clean > $O/local-fixture.log; echo fixture=${PIPESTATUS[0]}
$R bash -c 'echo "# design.md lint, offline (unshare -rn), pinned test-only deps (lint only; convert+build are exercised by local-fixture)"; rc=0; for f in input/fannan-review-pack/example-DESIGN.md test-results/local-fixture/design-output/test/fixture-DESIGN.md test-results/r5/OPUS-04-sample-DESIGN.md; do echo "\$ design.md lint $f"; unshare -rn node runtime-deps/node_modules/@google/design.md/dist/index.js lint $f | python3 -c "import sys,json; t=sys.stdin.read(); j=json.loads(t[t.index(\"{\"):]); s=j.get(\"summary\",j); print(json.dumps({k:s.get(k) for k in (\"errors\",\"warnings\",\"infos\")}))"; r=${PIPESTATUS[0]}; echo exit=$r; [ $r = 0 ] || rc=1; done; exit $rc' 2>&1 | clean > $O/designmd-lint-offline.log; echo lint=${PIPESTATUS[0]}
